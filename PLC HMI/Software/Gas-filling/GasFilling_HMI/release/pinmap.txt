[DISPLAY - ST7796 SPI TFT 480x320 @ 40MHz]
MOSI = GPIO10
MISO = GPIO11
SCLK = GPIO12
CS   = GPIO13
DC   = GPIO14
RST  = -1 (Connected to EN/Hardware Reset)

[TOUCH - Goodix GT911 Capacitive Touch I2C]
SDA = GPIO8
SCL = GPIO9
INT = GPIO40
RST = GPIO39
Mapping: X = 479 - raw_y, Y = raw_x

[DIGITAL INPUTS - Optocouplers]
I0 = کلیدهای تابع F1 تا F5 (کیپد پنل - GPIO1 ADC)
I1 = کلیدهای اصلی پنل (MODE / START / STOP / SAVE / CANCEL - GPIO2 ADC)
I2 = پدال استارت پایی اپراتور (GPIO7 - Active LOW optocoupler)
I3 = استپ اضطراری سخت‌افزاری (GPIO3 - Active LOW optocoupler)
I4 = تاچ اسکرین خازنی GT911 (I2C)
I5 = سنسور ورودی شیشه / آماده‌باش
M0 = ورودی/خروجی کمکی 0
M1 = ورودی/خروجی کمکی 1
M2 = ورودی/خروجی کمکی 2
M3 = ورودی/خروجی کمکی 3
M4 = ورودی/خروجی کمکی 4
M5 = ورودی/خروجی کمکی 5
M6 = ورودی/خروجی کمکی 6
M7 = ورودی/خروجی کمکی 7

[DIGITAL OUTPUTS - Relays]
Q0 = رله شیر برقی ۱ تزریق (شیشه ۲ - ایستگاه چپ / GPIO16 - Configurable N.O / N.C)
Q1 = رله شیر برقی ۲ تزریق (شیشه ۱ - ایستگاه راست / GPIO42 - Configurable N.O / N.C)
Q2 = خروجی کمکی رله ۲
Q3 = خروجی کمکی رله ۳
M0 = ورودی/خروجی کمکی 0
M1 = ورودی/خروجی کمکی 1

M7 = ورودی/خروجی کمکی 7
